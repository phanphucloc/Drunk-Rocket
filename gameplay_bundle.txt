  /**
   * NGUON SU THAT DUY NHAT cho tien trinh: tinh lai tu SO CONG da vuot.
   * Diem (near-miss / smash / hat gio) KHONG lam nhay chuong -> nhip thoi tiet on dinh.
   */
  _syncProgression(force = false) {
    const prog = getProgression(this.score);
    const prev = this._progState;
    this._progState = prog;

    this.currentTheme = prog.theme;
    this.kiteStage = prog.kiteStage;
    this.colorStage = prog.colorStage;
    this.lapNumber = prog.lapNumber;
    this.chapterNumber = prog.chapterNumber;
    this._chapterGlobal = prog.globalChapter;
    this.difficulty = difficultyFor(prog.lapIndex);
    if (this.pillars) this.pillars.tierBoost = prog.lapIndex * 2;

    const chapterChanged = force || !prev || prog.globalChapter !== prev.globalChapter;
    const lapChanged = force || !prev || prog.lapNumber !== prev.lapNumber;

    if (chapterChanged || lapChanged) {
      this.starfield.setTheme(this.currentTheme);
      if (this.weather) this.weather.setTheme(this.currentTheme);
      if (this.scenic) this.scenic.setTier(this.currentTheme.tier);
      if (this.audio && this.audio.setMusicTheme) this.audio.setMusicTheme(this.currentTheme.music);
      this.hud.triggerSmashEffects();
      this.invincibleTimer = Math.max(this.invincibleTimer, 1.2);
      if (infoPanels) infoPanels.setLandmark(this.currentTheme, prog.lapNumber, prog.chapterNumber, true,
        this.score % POINTS_PER_CHAPTER, POINTS_PER_CHAPTER);
    }
    if (lapChanged && this.audio && this.audio.playKiteUpgrade) {
      this.audio.playKiteUpgrade(prog.kiteStage);
    }
    return prog;
  }

  _onScore(pillar) {
    this.score += 1;
    this.justScoredTimer = 0.6;

    // CONG da vuot la nguon tien trinh (chuong/vong/mau dieu);
    // diem chi dung cho bang xep hang.
    this._syncProgression();
    if (infoPanels) infoPanels.setLandmark(this.currentTheme, this.lapNumber, this.chapterNumber, false,
      this.score % POINTS_PER_CHAPTER, POINTS_PER_CHAPTER);

    // Má»‘c má»Ÿ TÃšI BÃ™A tÃ­nh theo ÄIá»‚M
    if (this.perks) this.perks.checkMilestone(this.score);

    // Regular pillar gives +10% Overdrive (10 pillars cleared = 100% Smash ready)
    this.drunkMeter = Math.min(100, this.drunkMeter + 10);
    this.audio.playScore(); // Cheerful "HEYYY!" sound
    this.particles.emitScoreSparkles(this.rocket.x + 20, this.rocket.y);

    const pX = pillar ? pillar.x + (pillar.width || 52) / 2 : this.rocket.x + 20;
    const pY = pillar ? pillar.gapY : this.rocket.y;
    this.particles.emitEnergySiphon(pX, pY, this.width / 2, 126, 9);

    // Spawn 2 to 4 moving floating energy particles midway between the cleared pillar and upcoming pillar
    const upcomingPillars = this.pillars.pillars.filter(p => !p.passed && !p.isDestroyed && p.x > (pillar ? pillar.x : this.rocket.x));
    const nextPillar = upcomingPillars.length > 0 ? upcomingPillars[0] : null;

    let midX;
    if (nextPillar && pillar) {
      const clearedRight = pillar.x + (pillar.width || 52);
      midX = (clearedRight + nextPillar.x) / 2;
    } else if (pillar) {
      midX = pillar.x + (pillar.width || 52) + 160;
    } else {
      midX = this.rocket.x + 180;
    }

    const clusterCount = Math.floor(Math.random() * 3) + 2; // 2, 3, or 4 cells
    const midY = (pillar && nextPillar) ? (pillar.gapY + nextPillar.gapY) / 2 : (pillar ? pillar.gapY : 350);
    this.energyCells.spawnCluster(midX, midY, clusterCount);

    if (this.perks) this.perks.onPassPillar();

    if (this.ghosts && pillar) this.ghosts.spawnForPillar(pillar, this.score);
    if (this.score > this.bestScore) {
      this.bestScore = this.score;
      localStorage.setItem('drunk_rocket_best', this.bestScore.toString());
    }
  }

  _onNearMiss(pillar) {
    const bonus = 2;                 // Near-miss: +2 Ä‘iá»ƒm cá»‘ Ä‘á»‹nh
    this.score += bonus;
    this._syncProgression();
    if (infoPanels) infoPanels.setLandmark(this.currentTheme, this.lapNumber, this.chapterNumber, false,
      this.score % POINTS_PER_CHAPTER, POINTS_PER_CHAPTER);
    this.justScoredTimer = 0.8;
    this.drunkMeter = Math.min(100, this.drunkMeter + (12 * this.nearMissMultiplier));
    this.audio.playNearMiss();
    this.hud.showNearMiss(bonus);
    this.particles.emitNearMiss(this.rocket.x, this.rocket.y);

    const pX = pillar ? pillar.x + (pillar.width || 52) / 2 : this.rocket.x + 20;
    const pY = pillar ? pillar.gapY : this.rocket.y;
    this.particles.emitEnergySiphon(pX, pY, this.width / 2, 126, 14);

    if (this.perks) this.perks.checkMilestone(this.score);
    if (this.ghosts && pillar) this.ghosts.spawnForPillar(pillar, this.score);
    if (this.score > this.bestScore) {
      this.bestScore = this.score;
      localStorage.setItem('drunk_rocket_best', this.bestScore.toString());
    }
  }

  /** Vá»‡t nÃ³n sÃ¡ng trÆ°á»›c mÅ©i diá»u khi Ä‘ang Lá»™ng GiÃ³. */
  _drawSmashLance() {
    const ctx = this.ctx;
    const rx = this.rocket.x;
    const ry = this.rocket.y;
    ctx.save();
    ctx.globalCompositeOperation = 'lighter';
    const grad = ctx.createLinearGradient(rx, ry, rx + 120, ry);
    grad.addColorStop(0, 'rgba(255, 236, 170, 0.75)');
    grad.addColorStop(1, 'rgba(255, 170, 60, 0)');
    ctx.fillStyle = grad;
    ctx.beginPath();
    ctx.moveTo(rx + 10, ry - 3);
    ctx.lineTo(rx + 120, ry);
    ctx.lineTo(rx + 10, ry + 3);
    ctx.closePath();
    ctx.fill();
    ctx.restore();
  }

  _triggerSupernovaSmash(forcedChain = false) {
    if (this.isSmashActive) return;
    const isChain = (this.nextSmashChain >= 2) || forcedChain || (this.smashChainCount >= 2);
    if (!forcedChain && this.drunkMeter < 100) return;

    this.isSmashActive = true;
    this.drunkMeter = 0;
    this.originalRocketX = this.rocket.x;
    this.smashStartX = this.rocket.x;
    this.smashPillarsLeft = isChain ? 2 : 1;
    this.smashSpeedMul = 3.4;
    this.smashReturnTimer = 0;
    this.smashChainCount = isChain ? 2 : 1;
    if (isChain) this.nextSmashChain = 0;
    this.smashLunge = 96;

    // CHá»ˆ cá»™t PHÃA TRÆ¯á»šC má»›i lÃ  má»¥c tiÃªu (bug cÅ©: cá»™t vá»«a Ä‘i qua váº«n bá»‹ tÃ­nh -> ná»• sau lÆ°ng)
    const ahead = this.pillars.pillars
      .filter(p => !p.isDestroyed && p.x > this.rocket.x + this.rocket.radius)
      .sort((a, b) => a.x - b.x);

    const targets = ahead.slice(0, this.smashPillarsLeft);
    this.smashTargets = targets;           // chá»‰ nhá»¯ng cá»™t nÃ y má»›i Ä‘Æ°á»£c hÃºc (khÃ´ng cá»™t nÃ o khÃ¡c)
    const afterTarget = ahead[this.smashPillarsLeft] || null;

    // Thá»i lÆ°á»£ng = Ä‘iá»ƒm GIá»®A cá»™t vá»«a phÃ¡ vÃ  cá»™t káº¿ tiáº¿p (Ä‘Ãºng yÃªu cáº§u "dá»«ng á»Ÿ giá»¯a 2 cá»™t")
    const worldSpeed = Math.max(140, (this.pillars.speed || 200) * this.smashSpeedMul);
    const noseX = this.rocket.x + (this.smashLunge || 96) + this.rocket.radius;
    const tReach = (px) => Math.max(0.05, (px - noseX) / worldSpeed);

    let duration;
    if (targets.length === 0) {
      duration = 0.5;                       // khÃ´ng cÃ³ cá»™t phÃ­a trÆ°á»›c: chá»‰ lÆ°á»›t má»™t nhá»‹p
    } else {
      const tLast = tReach(targets[targets.length - 1].x);
      const tAfter = afterTarget ? tReach(afterTarget.x) : tLast + 0.45;
      duration = Math.max(0.4, Math.min(1.6, (tLast + tAfter) / 2));
    }

    this.smashTime = 0;
    this.smashDurationTotal = duration;
    this.smashTimeout = duration + 0.35;
    this.smashAlignY = targets[0] ? targets[0].gapY : this.rocket.y;

    this.audio.playSmash(isChain);
    this.hud.triggerSmashEffects();
    this.invincibleTimer = Math.max(this.invincibleTimer, duration + 0.2);
  }

  /** Káº¿t thÃºc cÃº Lá»™ng GiÃ³: háº¡ tá»‘c, trÃ´i vá» lÃ n bay, báº¥t tá»­ ngáº¯n + cooldown chá»‘ng smash liÃªn tiáº¿p. */
  _endSmash() {
    this.isSmashActive = false;
    this.smashTargets = [];
    // QoL: ngÆ°á»i chÆ¡i hay quÃªn báº¥m láº¡i sau cÃº lÆ°á»›t -> táº·ng 1 nhá»‹p Ä‘áº©y Ä‘á»ƒ khÃ´ng rÆ¡i ngay
    if (this.state === 'PLAYING' && this.rocket && !this.rocket.isDead) {
      this.rocket.thrust();
      if (this.audio && this.audio.playThrust) this.audio.playThrust();
      this.particles.emitThrust(this.rocket.x, this.rocket.y, this.rocket.rotation);
    }
    this.smashSpeedMul = 1;
    if (this.pillars) this.pillars.speedMul = 1;
    this.smashReturnTimer = 0.24;
    this.invincibleTimer = Math.max(this.invincibleTimer, 0.7);
    this.smashChainCount = 1;

  setPaused(paused) {
    const next = !!paused;
    if (this.paused === next) return;
    this.paused = next;

    if (this.input) {
      this.input.consumeTap();
      this.input.consumeDoubleTap();
    }
    if (this.audio) {
      if (next) {
        if (this.audio.stopLaserProximity) this.audio.stopLaserProximity();
        if (this.audio.setMusicDuck) this.audio.setMusicDuck(0.35);

    this.state = 'GAMEOVER';
    if (this.audio && this.audio.stopLaserProximity) this.audio.stopLaserProximity();
    this.timeSinceDeath = 0;

    this.particles.emitExplosion(safeX, safeY);

    const runDuration = performance.now() - this.runStartTime;
    if (this.ghosts) this.ghosts.submitGhost(localStorage.getItem('drunk_player_name') || 'ACE', this.score, safeY);
    const gameoverActions = document.getElementById('gameover-actions');
    if (gameoverActions) gameoverActions.classList.remove('hidden');

    const isRecord = this.score > 0 && this.score >= this.bestScore;
    if (isRecord && this.leaderboard) {

    } else {
      const act = document.getElementById('gameover-actions');
      if (act) act.classList.remove('hidden');
    }

      this.audio.updateMusicTick(dt);

    this.starfield.update(dt, this.state === 'PLAYING' ? 1 : 0.2);
    if (this.scenic) this.scenic.update(dt, this.currentTheme ? this.currentTheme.tier : 1);
    if (this.weather) this.weather.update(dt);

    if (this.invincibleTimer > 0) this.invincibleTimer = Math.max(0, this.invincibleTimer - dt);

    this._stampTick = (this._stampTick || 0) + dt;

      this.input.consumeDoubleTap();

        }

      }
    } else if (this.state === 'PLAYING') {
      if (this.smashCooldown > 0) this.smashCooldown = Math.max(0, this.smashCooldown - dt);
      if (hasSmash && this.drunkMeter >= 100 && (this.smashCooldown || 0) <= 0) {

      if (this.isSmashActive) {
        this.smashTimeout -= dt;
        this.smashTime += dt;

        // 1) CÃº tiáº¿n ngáº¯n (ease-out ~0.22s) â€” khÃ´ng giáº­t, khÃ´ng dá»«ng giá»¯a chá»«ng

        if (this.pillars) this.pillars.speedMul = this.smashSpeedMul;

        //    Nhá» chá»‘t theo Ä‘á»‘i tÆ°á»£ng, cá»™t vá»«a bay qua / cá»™t tá»± sinh khÃ´ng bao giá» bá»‹ ná»•.
        for (const pillar of (this.smashTargets || [])) {

            this.particles.emitPillarSmash(pcx, botColY);

            this._syncProgression();
            if (infoPanels) infoPanels.setLandmark(this.currentTheme, this.lapNumber, this.chapterNumber, false,

            this.smashPillarsLeft -= 1;
          }

        this._drawSmashLance();
        this.particles.emitSmashFireTrail(this.rocket.x, this.rocket.y, this.rocket.angle || 0);

      this.energyCells.update(dt, this.pillars.speed, this.rocket, this.hasDrunkMagnet, this._onCollectCell);

      // Proximity Laser Hum & Electric Crackle ("xáº¹t xáº¹t" gets louder as rocket approaches beam)
      if (!this.isSmashActive && this.quantumTimer <= 0) {
        let maxProximity = 0;
        for (const p of this.pillars.pillars) {
          if (p.isDestroyed) continue;
          if (p.x + p.width > this.rocket.x - 45 && p.x < this.rocket.x + 220) {
            const cx = p.x + p.width / 2;
            const cy = p.gapY;
            let rx = this.rocket.x;
            let ry = this.rocket.y;
            if (p.angle !== 0) {
              const cos = Math.cos(-p.angle);
              const sin = Math.sin(-p.angle);
              const dx = this.rocket.x - cx;
              const dy = this.rocket.y - cy;
              rx = cx + (dx * cos - dy * sin);
              ry = cy + (dx * sin + dy * cos);
            }

            // Distance to active vertical laser beam in the gap
            const clampedY = Math.max(p.topY, Math.min(p.bottomY, ry));
            const distToBeam = Math.hypot(rx - cx, ry - clampedY);

            // Distance to top/bottom emitter heads
            const distToTopHead = Math.hypot(rx - cx, ry - p.topY);
            const distToBottomHead = Math.hypot(rx - cx, ry - p.bottomY);
            const minDist = Math.min(distToBeam, distToTopHead, distToBottomHead);

            const dangerRadius = 180;
            if (minDist < dangerRadius) {
              const prox = Math.max(0, 1 - (minDist / dangerRadius));
              maxProximity = Math.max(maxProximity, prox);
            }
          }
        }
        if (this.audio && this.audio.updateLaserProximity) {
          this.audio.updateLaserProximity(maxProximity);
        }
      } else {
        if (this.audio && this.audio.stopLaserProximity) {
          this.audio.stopLaserProximity();
        }
      }

      // Boundary collision check (ceiling or floor)
      // TRáº¦N Má»€M: cháº¡m mÃ©p trÃªn thÃ¬ náº£y nháº¹ xuá»‘ng (tháº£ diá»u Ä‘á»¥ng táº§ng mÃ¢y),

      if (!this.isSmashActive && this.invincibleTimer <= 0) {

          this.rocket.vy = Math.max(this.rocket.vy, 140);

          this.ceilingTimer = 2.0;
          if (this.audio && this.audio.playThrust) this.audio.playThrust();
        }
        if (this.rocket.y + this.rocket.radius >= this.height) {
          this._die();
          return;
        }
      }
      if (this.ceilingTimer > 0) {
        this.ceilingTimer -= dt;
        if (this.ceilingTimer <= 0) this.ceilingHits = 0;
      }
      if ((this.ceilingHits || 0) >= 3) {
        this._die();
        return;
      }

      // Pillar collision check
      if (!this.isSmashActive && this.quantumTimer <= 0 && this.invincibleTimer <= 0) {
        const collided = this.pillars.checkCollision(this.rocket);
        if (collided) {
          if (this.shieldActive) {
            this.shieldActive = false;
            collided.isDestroyed = true;

            this.audio.playSmash();
            this.quantumTimer = 1.2;
            this.rocket.isQuantum = true;
          } else {
            this._die();
            return;
          }
        }
      }
    } else if (this.state === 'GAMEOVER') {
      this.timeSinceDeath += dt;
      if (this.timeSinceDeath >= 0.55 && hasTap) {
        this._restart();
      }
    }
  }

/**
 * Drunk Physics Engine
 * Implements asymmetrical gravity (down-left drift) and angled thrust (up-right impulse),
 * coupled with drunken angular inertia and wobble.
 */
export const PHYSICS_CONFIG = {
  // Gravity
  GRAVITY_Y: 1050,         // Vertical downward acceleration (px/s^2)
  GRAVITY_X: -140,         // Leftward drift acceleration (px/s^2)

  // Thrust Impulse
  THRUST_VY: -350,         // Upward velocity impulse on tap (px/s)
  THRUST_VX: 130,          // Rightward forward surge on tap (px/s)
  MAX_VY: 600,             // Terminal downward velocity
  MIN_VY: -450,            // Terminal upward velocity

  // Horizontal boundaries & damping
  BASE_X: 100,             // Natural equilibrium X position
  MIN_X: 45,               // Left screen boundary threshold
  MAX_X: 200,              // Right screen boundary threshold
  DRAG_X: 3.5,             // Horizontal aerodynamic damping

  // Angular kinematics
  THRUST_ANGLE: -0.60,     // ~ -35 degrees (nose up-right on thrust)
  FALL_ANGLE_MAX: 0.85,    // ~ +48 degrees (nose down-left when dropping)
  ROTATION_SPEED: 4.5,     // Speed of rotating toward target fall angle

  // Drunk wobble
  WOBBLE_FREQ: 5.5,        // Sinusoidal wobble frequency (rad/s)
  WOBBLE_AMP: 35,          // Wobble velocity amplitude (px/s)
};

export class PhysicsEngine {
  /**
   * Applies one physics update tick to the rocket.
   * @param {Rocket} rocket
   * @param {number} dt Delta time in seconds
   * @param {boolean} hasThrust Whether a tap thrust occurred this frame
   * @param {number} totalTime Game elapsed time in seconds
   */
  static updateRocket(rocket, dt, hasThrust, totalTime) {
    if (rocket.isDead) return;

    if (hasThrust) {
      // Apply angled thrust: upwards and slightly to the right
      rocket.vy = PHYSICS_CONFIG.THRUST_VY;
      rocket.vx = Math.min(rocket.vx + PHYSICS_CONFIG.THRUST_VX, 180);
      rocket.angle = PHYSICS_CONFIG.THRUST_ANGLE;
      rocket.isThrusting = true;
      rocket.thrustTimer = 0.12; // Visual flame duration
    } else {
      // Natural gravitational acceleration
      rocket.vy += PHYSICS_CONFIG.GRAVITY_Y * dt;
      rocket.vx += PHYSICS_CONFIG.GRAVITY_X * dt;

      // Add drunken wobble to horizontal velocity
      const wobble = Math.sin(totalTime * PHYSICS_CONFIG.WOBBLE_FREQ) * PHYSICS_CONFIG.WOBBLE_AMP;
      rocket.vx += wobble * dt;

      // Restoring force toward base X so rocket never gets permanently lost off-screen
      const displacement = rocket.x - PHYSICS_CONFIG.BASE_X;
      rocket.vx -= displacement * 1.8 * dt;

      // Apply horizontal drag
      rocket.vx -= rocket.vx * PHYSICS_CONFIG.DRAG_X * dt;

      // Update rotation (tilts downward as it falls)
      if (rocket.angle < PHYSICS_CONFIG.FALL_ANGLE_MAX) {
        rocket.angle += PHYSICS_CONFIG.ROTATION_SPEED * dt;
      }

      if (rocket.thrustTimer > 0) {
        rocket.thrustTimer -= dt;
        if (rocket.thrustTimer <= 0) {
          rocket.isThrusting = false;
        }
      }
    }

    // Clamp vertical velocity
    if (rocket.vy > PHYSICS_CONFIG.MAX_VY) rocket.vy = PHYSICS_CONFIG.MAX_VY;
    if (rocket.vy < PHYSICS_CONFIG.MIN_VY) rocket.vy = PHYSICS_CONFIG.MIN_VY;

    // Integrate positions
    rocket.x += rocket.vx * dt;
    rocket.y += rocket.vy * dt;

    // Clamp horizontal position inside playable lane
    if (rocket.x < PHYSICS_CONFIG.MIN_X) {
      rocket.x = PHYSICS_CONFIG.MIN_X;
      rocket.vx = 0;
    } else if (rocket.x > PHYSICS_CONFIG.MAX_X) {
      rocket.x = PHYSICS_CONFIG.MAX_X;
      rocket.vx = 0;
    }
  }

  /**
   * Circle vs AABB Collision Check (Pixel-perfect fair hitbox).
   * @param {number} cx Circle X
   * @param {number} cy Circle Y
   * @param {number} cr Circle Radius
   * @param {number} rx Box X
   * @param {number} ry Box Y
   * @param {number} rw Box Width
   * @param {number} rh Box Height
   * @returns {boolean}
   */
  static checkCircleAABB(cx, cy, cr, rx, ry, rw, rh) {
    const closestX = Math.max(rx, Math.min(cx, rx + rw));
    const closestY = Math.max(ry, Math.min(cy, ry + rh));

    const distanceX = cx - closestX;
    const distanceY = cy - closestY;

    return (distanceX * distanceX + distanceY * distanceY) < (cr * cr);
  }
}

        globalChapter: chapterIndex,
        kiteStage: LAPS[i].kite,
        colorStage: colorStageFor(remaining),

export class PerkDraftManager {
  constructor(game) {
    this.game = game;
    this.isDrafting = false;
    this.isPreDraft = false;
    this.preDraftTimer = 0;
    this.isPostDraft = false;
    this.postDraftTimer = 0;
    this.selectedCard = null;
    this.hoveredIndex = -1;
    this.draftCards = [];
    this.draftTimer = 0;
    this.draftMaxDuration = 2.4;
    this.lastMilestoneScore = -6; // má»‘c Ä‘áº§u á»Ÿ 6 Cá»”NG, sau Ä‘Ã³ 18, 30... (lá»‡ch pha vá»›i chÆ°Æ¡ng)
    this.milestoneInterval = 12;
  }

  reset() {
    this.isDrafting = false;
    this.isPreDraft = false;
    this.preDraftTimer = 0;
    this.isPostDraft = false;
    this.postDraftTimer = 0;
    this.selectedCard = null;
    this.hoveredIndex = -1;
    this.draftCards = [];
    this.draftTimer = 0;
    this.lastMilestoneScore = -6;
  }

  checkMilestone(currentScore) {
    // If rocket is currently dashing in Supernova Smash, wait until dash finishes
    if (this.game && this.game.isSmashActive) return;
    if (this.isDrafting || this.isPreDraft || this.isPostDraft) return;

    if (currentScore >= this.lastMilestoneScore + this.milestoneInterval) {
      this.lastMilestoneScore = currentScore;
      // 1.0s pre-draft pause before showing cards
      this.isPreDraft = true;
      this.preDraftTimer = 0.45;
      if (this.game && this.game.rocket) {
        this.game.rocket.vx = 0;
        this.game.rocket.vy = 0;
      }
      if (this.game && this.game.audio) {
        this.game.audio.playMedal();
      }
    }
  }

  triggerDraft() {
    // REQUIREMENT: Má»—i tháº» bÃ i xuáº¥t hiá»‡n nhiá»u láº§n vÃ  ngáº«u nhiÃªn (khÃ´ng bá»‹ cháº·n vÄ©nh viá»…n)
    const shuffled = [...PERK_CATALOG].sort(() => 0.5 - Math.random());
    this.draftCards = shuffled.slice(0, 3);
    this.isDrafting = true;
    this.hoveredIndex = -1;
    this.draftTimer = this.draftMaxDuration;

    if (this.game && this.game.audio) {
      this.game.audio.playScore();
    }
  }

  setHover(clientX, clientY) {
    if (!this.isDrafting || this.draftCards.length === 0) {
      this.hoveredIndex = -1;
      return;
    }

    const cardW = 320;
    const cardH = 75;
    const cardX = this.game.width / 2 - cardW / 2;
    const startY = 240;

    let found = -1;
    if (clientX >= cardX && clientX <= cardX + cardW) {
      for (let idx = 0; idx < this.draftCards.length; idx++) {
        const cardY = startY + idx * 95;
        if (clientY >= cardY && clientY <= cardY + cardH) {
          found = idx;
          break;
        }
      }
    }
    this.hoveredIndex = found;
  }

  selectCard(index) {
    if (!this.isDrafting || index < 0 || index >= this.draftCards.length) return;
    const perk = this.draftCards[index];

    this.selectedCard = perk;
    this.isDrafting = false;
    this.hoveredIndex = -1;

    // Apply or Reset perk duration back to full
    perk.apply(this.game);

    // Consume all inputs immediately
    if (this.game && this.game.input) {
      this.game.input.consumeTap();
      this.game.input.consumeDoubleTap();
      this.game.perkDraftCooldown = 0.5;
    }
    if (this.game && this.game.audio) {
      this.game.audio.playMedal();
    }

    if (perk.id === 'chain_reaction') {

      this.isPostDraft = false;
      this.selectedCard = null;
      this.draftCards = [];
      this.game.invincibleTimer = Math.max(this.game.invincibleTimer, 0.8);
    } else {
      // CÃ¡c tháº» khÃ¡c: 1.0s post-draft freeze pause Ä‘á»ƒ ngÆ°á»i chÆ¡i chuáº©n bá»‹
      this.isPostDraft = true;
      this.postDraftTimer = 0.7;
    }
  }

  onPassPillar() {
    if (!this.game || !this.game.perkDurations) return;

    for (const perk of PERK_CATALOG) {
      if (this.game.perkDurations[perk.id] !== undefined) {
        this.game.perkDurations[perk.id]--;
        // Khi Ä‘i qua Ä‘á»§ sá»‘ cá»™t -> háº¿t hiá»‡u lá»±c, gá»i remove()
        if (this.game.perkDurations[perk.id] <= 0) {
          perk.remove(this.game);
        }
      }
    }
  }

  update(dt) {
    // 1. Pre-draft 1.0s freeze
    if (this.isPreDraft) {
      this.preDraftTimer -= dt;
      if (this.preDraftTimer <= 0) {
        this.isPreDraft = false;
        this.triggerDraft();
      }
      return;
    }

    // 2. Card Drafting
    if (this.isDrafting) {
      this.draftTimer -= dt;
      if (this.draftTimer <= 0) {
        this.selectCard(0);
      }
      return;
    }

    // 3. Post-draft 1.0s ready pause
    if (this.isPostDraft) {
      this.postDraftTimer -= dt;
      if (this.postDraftTimer <= 0) {
        this.isPostDraft = false;
        this.selectedCard = null;
        this.draftCards = [];
      }
    }
  }

/**
 * Input Manager for Drunk Rocket 2.0
 * Handles Responsive Tap (Thrust) and Double Tap (Supernova Smash)
 */
export class InputManager {
  constructor(targetElement = window) {
    this.targetElement = targetElement;
    this.justPressed = false;
    this.doubleTapped = false;      // giá»¯ cho tÆ°Æ¡ng thÃ­ch (khÃ´ng cÃ²n dÃ¹ng Ä‘á»ƒ Smash)
    this.isPressed = false;
    this.lastTapTime = 0;
    this.doubleTapThreshold = 280;

    // Lá»™ng GiÃ³ báº±ng NHáº¤N GIá»®: giá»¯ >= HOLD_MS thÃ¬ kÃ­ch hoáº¡t (tap thÆ°á»ng ~50-110ms)
    this.HOLD_MS = 200;
    this.holdStart = 0;
    this.holdProgress = 0;          // 0..1 Ä‘á»ƒ váº½ vÃ²ng náº¡p quanh diá»u
    this.smashRequested = false;
    this._holdArmed = false;

    this._onKeyDown = this._onKeyDown.bind(this);
    this._onKeyUp = this._onKeyUp.bind(this);
    this._onPointerDown = this._onPointerDown.bind(this);
    this._onPointerUp = this._onPointerUp.bind(this);

    this._bindEvents();
  }

  _bindEvents() {
    window.addEventListener('keydown', this._onKeyDown, { passive: false });
    window.addEventListener('keyup', this._onKeyUp, { passive: false });

    this.targetElement.addEventListener('pointerdown', this._onPointerDown, { passive: false });
    window.addEventListener('pointerup', this._onPointerUp, { passive: false });
    window.addEventListener('pointercancel', this._onPointerUp, { passive: false });
  }

  consumeSmash() {
    if (!this.smashRequested) return false;
    this.smashRequested = false;
    return true;
  }

  /** Cáº­p nháº­t tiáº¿n trÃ¬nh giá»¯ tay (gá»i má»—i frame). */
  updateHold(dt) {
    if (!this.isPressed || !this._holdArmed) {
      this.holdProgress = 0;
      return;
    }
    const held = performance.now() - this.holdStart;
    this.holdProgress = Math.max(0, Math.min(1, held / this.HOLD_MS));
    if (held >= this.HOLD_MS) {
      this.smashRequested = true;
      this.holdProgress = 1;
      this._holdArmed = false;      // chá»‰ kÃ­ch hoáº¡t 1 láº§n cho má»—i láº§n giá»¯
    }
  }

  _handleTap() {
    const now = performance.now();
    const timeDiff = now - this.lastTapTime;

    // LuÃ´n ghi nháº­n lÃ  1 cÃº tap Ä‘á»ƒ Ä‘áº©y tÃ u bay lÃªn, khÃ´ng bao giá» bá»‹ nuá»‘t phÃ­m
    this.justPressed = true;

    // Náº¿u khoáº£ng cÃ¡ch giá»¯a 2 láº§n báº¥m tá»« 20ms Ä‘áº¿n 280ms -> ghi nháº­n thÃªm Double Tap
    if (timeDiff > 20 && timeDiff <= this.doubleTapThreshold) {
      this.doubleTapped = true;
    }
    this.lastTapTime = now;
  }

  _onKeyDown(e) {
    if (e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'KeyW') {
      e.preventDefault();
      // NgÄƒn keydown auto-repeat cá»§a há»‡ Ä‘iá»u hÃ nh lÃ m spam phÃ­m, nhÆ°ng cho phÃ©p nháº¥p nháº£ nhanh
      if (!e.repeat) {
        this._handleTap();
        // Giá»¯ Space/W Ä‘á»ƒ Lá»™ng GiÃ³ (giá»‘ng nháº¥n giá»¯ trÃªn mobile)
        this.isPressed = true;
        this.holdStart = performance.now();
        this._holdArmed = true;
      }
    }
  }

  _onKeyUp(e) {
    if (e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'KeyW') {
      this.isPressed = false;
      this._holdArmed = false;
      this.holdProgress = 0;
    }
  }

  _onPointerDown(e) {
    this.isPressed = true;
    this.holdStart = performance.now();
    this._holdArmed = true;

    e.preventDefault();
    this._handleTap();
    this.isPressed = true;
  }

  _onPointerUp(e) {
    this.isPressed = false;
    this._holdArmed = false;
    this.holdProgress = 0;

    this.isPressed = false;
  }

  consumeTap() {
    if (this.justPressed) {
      this.justPressed = false;
      return true;
    }
    return false;
  }

  consumeDoubleTap() {
    if (this.doubleTapped) {
      this.doubleTapped = false;
      return true;
    }
    return false;
  }

  isDown() {
    return this.isPressed;
  }

  destroy() {
    window.removeEventListener('keydown', this._onKeyDown);
    window.removeEventListener('keyup', this._onKeyUp);
    this.targetElement.removeEventListener('pointerdown', this._onPointerDown);
    window.removeEventListener('pointerup', this._onPointerUp);
    window.removeEventListener('pointercancel', this._onPointerUp);
  }
}

import { PhysicsEngine } from '../core/physics.js';
import { getThemeForScore, getTierFromScore } from '../core/progression.js';

export class LaserPillar {
  constructor(x, gapY, gapHeight, width = 52, options = {}) {
    this.x = x;
    this.baseGapY = gapY;
    this.gapY = gapY;
    this.gapHeight = gapHeight;
    this.width = width;
    this.passed = false;
    this.nearMissChecked = false;
    this.isDestroyed = false;
    this.quantumPhased = false;
    this.emitterHeight = 24;
    this.pulsePhase = Math.random() * Math.PI * 2;

    // Visual theme
    this.theme = options.theme || null;
    this.tier = options.tier || 1;

    // Dynamic difficulty movement (oscillating gapY & tilted/swaying angle)
    this.oscAmp = options.oscAmp || 0;
    this.oscFreq = options.oscFreq || 0;
    this.oscPhase = options.oscPhase !== undefined ? options.oscPhase : Math.random() * Math.PI * 2;

    this.baseAngle = options.baseAngle || 0;
    this.angleAmp = options.angleAmp || 0;
    this.angleFreq = options.angleFreq || 0;
    this.anglePhase = options.anglePhase !== undefined ? options.anglePhase : Math.random() * Math.PI * 2;
    this.angle = this.baseAngle;
  }

  get topY() {
    return this.gapY - this.gapHeight / 2;
  }

  get bottomY() {
    return this.gapY + this.gapHeight / 2;
  }

  update(dt, speed) {
    this.x -= speed * dt;
    this.pulsePhase += dt * 8;

    // Vertical bobbing oscillation
    if (this.oscAmp > 0) {
      this.oscPhase += dt * this.oscFreq;
      const targetY = this.baseGapY + Math.sin(this.oscPhase) * this.oscAmp;
      this.gapY = Math.max(130 + this.gapHeight / 2, Math.min(570 - this.gapHeight / 2, targetY));
    }

    // Tilted angular oscillation / sway (leaning left <-> right)
    if (this.angleAmp > 0) {
      this.anglePhase += dt * this.angleFreq;
      this.angle = this.baseAngle + Math.sin(this.anglePhase) * this.angleAmp;
    } else {
      this.angle = this.baseAngle;
    }
  }

  isOffscreen() {
    return this.x + this.width < -70;
  }

  /**
   * Check Near-Miss: Rocket passes very close (<16px margin) to top or bottom hazard
   * Coordinates are transformed into local unrotated pillar space for exact detection.
   */
  checkNearMiss(rocket) {
    if (this.isDestroyed || this.nearMissChecked) return false;

    const cx = this.x + this.width / 2;
    const cy = this.gapY;
    let rx = rocket.x;
    let ry = rocket.y;

    if (this.angle !== 0) {
      const cos = Math.cos(-this.angle);
      const sin = Math.sin(-this.angle);
      const dx = rocket.x - cx;
      const dy = rocket.y - cy;
      rx = cx + (dx * cos - dy * sin);
      ry = cy + (dx * sin + dy * cos);
    }

    // Check when rocket is horizontally passing through the pillar
    const inXRange = rx >= this.x - 10 && rx <= this.x + this.width + 10;
    if (inXRange) {
      const distToTop = (ry - rocket.radius) - this.topY;
      const distToBottom = this.bottomY - (ry + rocket.radius);

      // Within close proximity but without colliding
      const nearMargin = 16;
      if ((distToTop > 0 && distToTop <= nearMargin) || (distToBottom > 0 && distToBottom <= nearMargin)) {
        this.nearMissChecked = true;
        return true;
      }
    }
    return false;
  }

  /**
   * Check if rocket has COMPLETELY passed through the pillar including the tail
   * Rocket nose is at x + radius, tail is at x - radius - 14 (nozzle/flame edge)
   */
  hasPassedCompletely(rocket) {
    const tailX = rocket.x - (rocket.radius + 14);
    const halfW = this.width / 2;
    const cx = this.x + halfW;
    const maxRightX = cx + halfW + Math.abs(Math.sin(this.angle || 0)) * 60;
    return tailX > maxRightX;
  }

  /**
   * Collision check against rocket (Pixel-perfect transformed AABB detection).
   */
  collidesWith(rocket) {
    if (this.isDestroyed || this.quantumPhased) return false;

    const cx = this.x + this.width / 2;
    const cy = this.gapY;
    let rx = rocket.x;
    let ry = rocket.y;

    if (this.angle !== 0) {
      const cos = Math.cos(-this.angle);
      const sin = Math.sin(-this.angle);
      const dx = rocket.x - cx;
      const dy = rocket.y - cy;
      rx = cx + (dx * cos - dy * sin);
      ry = cy + (dx * sin + dy * cos);
    }

    // 1. Top pillar (from ceiling above y=0 down to this.topY)
    if (PhysicsEngine.checkCircleAABB(
      rx, ry, rocket.radius,
      this.x, -600, this.width, this.topY + 600
    )) {
      return true;
    }

    // 2. Bottom pillar (from y=this.bottomY down past screen bottom)
    if (PhysicsEngine.checkCircleAABB(
      rx, ry, rocket.radius,
      this.x, this.bottomY, this.width, 1600
    )) {
      return true;
    }

    return false;
  }
}

    this.speed = this.baseSpeed + Math.min(Math.floor(score / 5) * 3, 24);

      if (typeof difficulty.breatherPause === 'number') this.breatherPause = difficulty.breatherPause;
    }

      // Nhá»‹p "thá»Ÿ": cá»© N cá»•ng thÃ¬ nghá»‰ thÃªm 1 nhá»‹p cho máº¯t Ä‘á»¡ má»i
      const rest = (this.breatherEvery > 0 && this.spawnedCount % this.breatherEvery === 0)
        ? this.breatherPause
        : 0;
      this.spawnTimer = this.spawnInterval + rest;

    const moveSpeed = this.speed * (this.speedMul || 1);
    for (let i = this.pillars.length - 1; i >= 0; i--) {

      }

      baseAngle = (Math.random() < 0.5 ? -1 : 1) * (0.05 + Math.random() * 0.03);

    const destroyed = [];
    let candidates = this.pillars
      .filter(p => !p.isDestroyed && p.x + p.width > rocketX - 25)
      .sort((a, b) => a.x - b.x);

    const theme = getThemeForScore(this.currentScore);
    const tier = getTierFromScore(this.currentScore);

    while (candidates.length < count) {
      let nextX;
      let nextGapY;

      if (candidates.length === 0) {
        nextX = Math.max(rocketX + 110, 200);
        nextGapY = this.minGapY + Math.random() * (this.maxGapY - this.minGapY);
      } else {
        const prevPillar = candidates[candidates.length - 1];
        nextX = Math.max(prevPillar.x + 120, prevPillar.x < 220 ? 320 : prevPillar.x + 140);
        nextGapY = Math.min(this.maxGapY, Math.max(this.minGapY, prevPillar.gapY + (Math.random() - 0.5) * 60));
      }

      const effectiveTier = Math.min(9, (tier || 1) + (this.tierBoost || 0));
      const newPillar = new LaserPillar(nextX, nextGapY, this.gapHeight, 52, { theme, tier: effectiveTier });
      this.pillars.push(newPillar);
      candidates.push(newPillar);
      candidates.sort((a, b) => a.x - b.x);
    }

    for (let i = 0; i < count; i++) {
      const p = candidates[i];
      p.isDestroyed = true;
      p.passed = true;
      destroyed.push(p);
    }

    this.spawnTimer = this.spawnInterval + (count >= 2 ? 0.9 : 0.4);
    return destroyed;
  }

/**
 * Floating Energy Cells / Plasma Crystals for Drunk Rocket 2.0
 * Swarms of 2-4 playful moving orbs floating through space.
 * Only charges Overdrive energy (+4% each) toward Smash without accelerating level score!
 */
export class EnergyCell {
  constructor(x, y, vx = -60, vy = 0) {
    this.x = x;
    this.y = y;
    this.baseY = y;
    this.vx = vx;
    this.vy = vy;
    this.radius = 7;
    this.pulsePhase = Math.random() * Math.PI * 2;
    this.floatSpeed = 2.5 + Math.random() * 3.5;
    this.floatAmp = 12 + Math.random() * 16;
    this.isCollected = false;
    this.isMagnetized = false;
  }

  update(dt, speed, rocket, hasMagnet = false) {
    this.pulsePhase += dt * this.floatSpeed;

    if (hasMagnet && !this.isCollected) {
      const dx = rocket.x - this.x;
      const dy = rocket.y - this.y;
      const dist = Math.hypot(dx, dy);
      const magnetRange = 220; // Broad magnet suction range

      if (dist < magnetRange) {
        this.isMagnetized = true;
        const pullForce = 950 * (1 - dist / magnetRange) + 420;
        this.vx += (dx / dist) * pullForce * dt;
        this.vy += (dy / dist) * pullForce * dt;
        this.x += this.vx * dt;
        this.y += this.vy * dt;
        return;
      }
    }

    this.isMagnetized = false;
    // Dynamic floating motion (drifts left and bobs up and down)
    this.x -= speed * dt;
    this.y = this.baseY + Math.sin(this.pulsePhase) * this.floatAmp;
  }

  isOffscreen() {
    return this.x + this.radius < -40;
  }

  checkCollision(rocket) {
    if (this.isCollected) return false;
    const dx = rocket.x - this.x;
    const dy = rocket.y - this.y;
    const dist = Math.hypot(dx, dy);
    return dist < (rocket.radius + this.radius + 6);
  }
}
